   273→        return;
   274→      }

   276→      if (data.user) {

   278→          setLoginError('incorrect password or username');

   280→          return;
   281→        }

   285→      }
   286→    } catch (err) {
   287→      console.log('Backend login unreachable:', err);

   290→      setIsSubmitting(false);
   291→    }

   299→    const rawPhone = regPhone.trim();

   318→      submittedAt: new Date().toISOString(),

   332→        method: 'POST',
   333→        headers: { 'Content-Type': 'application/json' },
   334→        body: JSON.stringify({
   335→          ...newUser,
   336→          password: regPassword,
   337→        }),

   342→        setRegError(data.error || 'Failed to register account.');
   343→        setIsSubmitting(false);
   344→        return;
   345→      }
   346→
   347→      if (data.user && data.user.id) {
   348→        registeredUser = data.user;
   349→      }
   350→    } catch (err) {
   351→      console.log('Backend signup notice:', err);
   352→    } finally {
   353→      setIsSubmitting(false);
   354→    }
   355→
   356→    saveRegisteredUser(registeredUser);
   357→    onLoginSuccess(registeredUser);
   358→  };

   360→  const handleRegister = async (e: React.FormEvent) => {
   361→    e.preventDefault();
   362→    setRegError(null);
   363→
   364→    const trimmedName = regName.trim();
   365→    const rawPhone = regPhone.trim();
   366→    const rawEmail = rawEmailSafe(regEmail.trim());
   367→    const normName = trimmedName.toLowerCase();
   368→    const normPhone = normalizePhone(rawPhone);
   369→    const normEmail = rawEmail.toLowerCase();

   372→      setRegError('Please enter your full name.');
   373→      return;
   374→    }
   375→    if (!rawPhone) {
   376→      setRegError('Please enter your active Mobile No.');
   377→      return;
   378→    }
   379→    if (!regPassword.trim()) {
   380→      setRegError('Please choose a password.');
   381→      return;
   382→    }
   383→    if (selectedRole === 'farmer' && !regFarmName.trim()) {
   384→      setRegError('Please enter your Farm name.');
   385→      return;
   386→    }
   387→
   388→    // MANDATORY VERIFICATION FIELD (both roles): a Valid Government ID photo

   390→    if (!regValidIdPreview) {
   391→      setRegError('A Valid Government ID photo is required — upload a clear .jpg/.jpeg/.png image (max 5MB) to submit your registration.');
   392→      return;
   393→    }
   394→
   395→    // Farmer role-specific: RSBSA Number is OPTIONAL — the format is only
   396→    // validated when the farmer actually types something in.
   397→    if (selectedRole === 'farmer' && regRsbsa.trim() && !isValidRsbsaNumber(regRsbsa)) {
   398→      setRegError(`The RSBSA Number looks invalid. ${RSBSA_FORMAT_HINT}, or leave the field empty.`);

   401→
   402→    // Local Duplicate Validations across all existing users

   404→
   405→    // Check duplicate name

   408→      setRegError(`An account with the name "${trimmedName}" is already registered. Please use a unique full name.`);

   410→    }
   411→
   412→    // Check duplicate phone

   414→      const dupPhone = existingUsers.find((u) => u.phone && normalizePhone(u.phone) === normPhone);
   415→      if (dupPhone) {
   416→        setRegError(`An account with the mobile number "${rawPhone}" is already registered.`);
   417→        return;
   418→      }
   419→    }
   420→
   421→    // Check duplicate email
   422→    if (normEmail) {
   423→      const dupEmail = existingUsers.find((u) => u.email && u.email.trim().toLowerCase() === normEmail);
   424→      if (dupEmail) {
   425→        setRegError(`An account with the email "${rawEmail}" is already registered.`);
   426→        return;
   427→      }
   428→    }

   430→    // REQUIREMENT: the user must set their location on the map FIRST.

   437→      setRegError('Please pin your exact location on the map to create your account.');

   439→    }
